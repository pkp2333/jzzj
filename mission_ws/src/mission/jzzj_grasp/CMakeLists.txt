cmake_minimum_required(VERSION 3.0.2)
project(jzzj_grasp)
find_package(catkin REQUIRED COMPONENTS rospy roscpp actionlib control_msgs trajectory_msgs geometry_msgs sensor_msgs std_msgs jzzj_grasp_interfaces kdl_parser urdf)
find_package(orocos_kdl REQUIRED)
catkin_python_setup()
catkin_package(CATKIN_DEPENDS jzzj_grasp_interfaces)
option(STAGE_C_BUILD_KINEMATICS "Build ARM64 MyArm adapter on target only" OFF)
if(STAGE_C_BUILD_KINEMATICS)
  execute_process(COMMAND rospack find arm OUTPUT_VARIABLE ARM_PACKAGE OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE ARM_FOUND)
  if(NOT ARM_FOUND EQUAL 0)
    message(FATAL_ERROR "arm package unavailable")
  endif()
  add_executable(stage_c_kinematics src/kinematics_node.cpp)
  target_compile_features(stage_c_kinematics PRIVATE cxx_std_11)
  target_include_directories(stage_c_kinematics PRIVATE ${catkin_INCLUDE_DIRS} ${orocos_kdl_INCLUDE_DIRS} ${ARM_PACKAGE}/include)
  target_link_libraries(stage_c_kinematics ${catkin_LIBRARIES} ${orocos_kdl_LIBRARIES} ${ARM_PACKAGE}/include/arm/libarm_kinemarics.so)
  add_dependencies(stage_c_kinematics ${catkin_EXPORTED_TARGETS})
  install(TARGETS stage_c_kinematics RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
endif()
catkin_install_python(PROGRAMS scripts/stage_c_server.py scripts/stage_c_client.py scripts/sim_context.py scripts/check_model.py scripts/check_fruit6.py scripts/run_verified_fruit6.py scripts/model_probe_overlay.py scripts/publish_fruit6_sample.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch params resources DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md MODEL_FRUIT6.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

if(CATKIN_ENABLE_TESTING)
  find_package(PythonInterp 3 REQUIRED)
  add_test(NAME stage_c_offline COMMAND ${PYTHON_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_stage_c.py)
  add_test(NAME stage_c_fruit6 COMMAND ${PYTHON_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fruit6.py)
  add_test(NAME stage_c_ros_contracts COMMAND ${PYTHON_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ros_contracts.py)
endif()
